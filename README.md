#include <iostream>
using namespace std;

int main() {
    int num1, num2;

    cout << "Enter first number: ";
    cin >> num1;

    cout << "Enter second number: ";
    cin >> num2;

    cout << "\nAddition: " << num1 + num2;
    cout << "\nSubtraction: " << num1 - num2;
    cout << "\nMultiplication: " << num1 * num2;
    cout << "\nDivision: " << num1 / num2;
    cout << "\nModulus: " << num1 % num2;

    return 0;
}
